cmake_minimum_required(VERSION 3.22)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS")
project(SalahCat VERSION 2.5.0 LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/llama.cpp EXCLUDE_FROM_ALL)
add_executable(SalahCat MACOSX_BUNDLE main.m MosqueFeature.m CompanionCore.m CompanionUI.m DesktopCat.m LocalEngine.mm)
target_compile_options(SalahCat PRIVATE -fobjc-arc -Wno-incompatible-pointer-types -Wno-deprecated-declarations)
target_link_libraries(SalahCat PRIVATE llama "-framework Cocoa" "-framework UserNotifications" "-framework Security" "-framework MapKit" "-framework CoreLocation" "-framework QuartzCore" "-framework CoreGraphics")
set_target_properties(SalahCat PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist" OUTPUT_NAME "SalahCat")
add_custom_command(TARGET SalahCat POST_BUILD
 COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/Assets/SalahCat" "$<TARGET_BUNDLE_CONTENT_DIR:SalahCat>/Resources/SalahCat"
 COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Assets/AppIcon.icns" "$<TARGET_BUNDLE_CONTENT_DIR:SalahCat>/Resources/"
 COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/Models.json" "${CMAKE_CURRENT_SOURCE_DIR}/NewsSources.json" "${CMAKE_CURRENT_SOURCE_DIR}/PrivacyInfo.xcprivacy" "${CMAKE_CURRENT_SOURCE_DIR}/Privacy-and-Licenses.txt" "${CMAKE_CURRENT_SOURCE_DIR}/Qwen-LICENSE.txt" "$<TARGET_BUNDLE_CONTENT_DIR:SalahCat>/Resources/"
 COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_SOURCE_DIR}/vendor/llama.cpp/LICENSE" "$<TARGET_BUNDLE_CONTENT_DIR:SalahCat>/Resources/llama-LICENSE.txt")
option(SC_BUILD_TESTS "Build standalone local tests" OFF)
if(SC_BUILD_TESTS)
 add_executable(CoreTests tests/CoreTests.m CompanionCore.m)
 target_include_directories(CoreTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_options(CoreTests PRIVATE -fobjc-arc -Wno-incompatible-pointer-types)
 target_link_libraries(CoreTests PRIVATE "-framework Foundation")
 add_executable(EngineTests tests/EngineTests.mm LocalEngine.mm)
 target_include_directories(EngineTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_options(EngineTests PRIVATE -fobjc-arc -Wno-incompatible-pointer-types -Wno-deprecated-declarations)
 target_link_libraries(EngineTests PRIVATE llama "-framework Foundation")
endif()

set(SC_BUNDLED_MODEL_PATH "" CACHE FILEPATH "Verified Qwen3-0.6B-Q8_0.gguf for offline-first distribution")
if(SC_BUNDLED_MODEL_PATH)
 file(SHA256 "${SC_BUNDLED_MODEL_PATH}" SC_MODEL_HASH)
 if(NOT SC_MODEL_HASH STREQUAL "9465e63a22add5354d9bb4b99e90117043c7124007664907259bd16d043bb031")
  message(FATAL_ERROR "Bundled model SHA-256 mismatch")
 endif()
 add_custom_command(TARGET SalahCat POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_BUNDLE_CONTENT_DIR:SalahCat>/Resources/Models" COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SC_BUNDLED_MODEL_PATH}" "$<TARGET_BUNDLE_CONTENT_DIR:SalahCat>/Resources/Models/Qwen3-0.6B-Q8_0.gguf")
endif()
if(SC_BUILD_TESTS)
 add_executable(UIRenderTests tests/UIRenderTests.m CompanionUI.m CompanionCore.m LocalEngine.mm)
 target_include_directories(UIRenderTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_options(UIRenderTests PRIVATE -fobjc-arc -Wno-incompatible-pointer-types -Wno-deprecated-declarations)
 target_link_libraries(UIRenderTests PRIVATE llama "-framework Cocoa")
endif()

set_property(TARGET SalahCat APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Models.json;${CMAKE_CURRENT_SOURCE_DIR}/NewsSources.json;${CMAKE_CURRENT_SOURCE_DIR}/PrivacyInfo.xcprivacy;${CMAKE_CURRENT_SOURCE_DIR}/Privacy-and-Licenses.txt")
set_target_properties(SalahCat PROPERTIES XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/AppStore.entitlements" XCODE_ATTRIBUTE_ENABLE_APP_SANDBOX "YES")
